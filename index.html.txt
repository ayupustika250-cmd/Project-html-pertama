Berikut adalah file HTML lengkap index.html aplikasi MSC (Monitoring Spot Check MSC) yang sudah diperbarui dengan desain grafik interaktif yang jauh lebih modern, rapi, dan informatif.
Pembaruan Visual Grafik:
 * Grafik Khusus per Business Unit (BU):
   * Menggunakan warna gradient modern dengan sudut batang yang melengkung (rounded bars).
   * Menampilkan angka total laporan secara presisi di atas setiap batang grafik.
 * Grafik Gabungan Indikator Lengkap:
   * Dibuat menjadi Multi-Axis Interactive Chart yang menggabungkan:
     * Bar Chart (Batang): Distribusi Spasing Terisi (2m, 2.5m, 3m) & Top Kontraktor.
     * Line Chart (Garis Tren): Rata-rata Depth (cm) & Penetrometer.
 * Fitur Unduh Gambar (PNG):
   * Tombol Unduh Gambar di setiap grafik untuk menyimpan gambar beresolusi tinggi langsung ke perangkat.

    function toggleAuth(type) {
      if (type === 'register') {
        document.getElementById('loginBox').classList.add('hidden');

    function handleLogin(e) {
      e.preventDefault();
      const userVal = document.getElementById('loginUsername').value.trim();
      const passVal = document.getElementById('loginPassword').value.trim();
      const alertBox = document.getElementById('loginAlert');

      const foundUser = users.find(u => u.username.toLowerCase() === userVal.toLowerCase() && u.password === passVal);

      if (!foundUser) {
        alertBox.textContent = 'Username atau Password salah!';
        alertBox.classList.remove('hidden');
        return;
      }

      if (foundUser.status === 'pending') {
        alertBox.textContent = 'Akun Anda masih menunggu persetujuan dari Admin.';
        alertBox.classList.remove('hidden');
        return;
      }

      if (foundUser.status === 'rejected') {
        alertBox.textContent = 'Permohonan akun Anda telah ditolak oleh Admin.';
        alertBox.classList.remove('hidden');
        return;
      }

      currentUser = foundUser;
      sessionStorage.setItem(STORAGE_KEY_SESSION, JSON.stringify(currentUser));
      alertBox.classList.add('hidden');
      checkSession();
    }

    function handleRegister(e) {
      e.preventDefault();
      const fullName = document.getElementById('regFullName').value.trim();
      const username = document.getElementById('regUsername').value.trim();
      const password = document.getElementById('regPassword').value.trim();
      const alertBox = document.getElementById('registerAlert');

      if (users.some(u => u.username.toLowerCase() === username.toLowerCase())) {
        alertBox.textContent = 'Username sudah digunakan, pilih username lain.';
        alertBox.className = 'mb-4 p-3 bg-red-100 text-red-700 rounded-lg text-xs font-medium';
        alertBox.classList.remove('hidden');
        return;
      }

      const newUser = {
        id: 'usr-' + Date.now(),
        fullName,
        username,
        password,
        role: 'user',
        status: 'pending'
      };

      users.push(newUser);
      saveUsersToStorage();

      alertBox.textContent = 'Pendaftaran berhasil! Silakan tunggu persetujuan dari Admin.';
      alertBox.className = 'mb-4 p-3 bg-green-100 text-green-700 rounded-lg text-xs font-medium';
      alertBox.classList.remove('hidden');
      e.target.reset();
    }

    function checkSession() {
      if (currentUser) {

        document.getElementById('userInfoHeader').classList.remove('hidden');

        document.getElementById('inpPelapor').value = displayFullName;

        // Apply RBAC UI Restrictions

          document.querySelectorAll('.admin-only').forEach(el => el.classList.remove('hidden'));

    function renderCharts() {
      const startDate = document.getElementById('filterStartDate').value;
      const endDate = document.getElementById('filterEndDate').value;
      const filterBU = document.getElementById('filterBU').value;
      const contractor = document.getElementById('filterContractor').value;

      let filtered = [...reports];

      if (startDate) filtered = filtered.filter(r => r.date >= startDate);
      if (endDate) filtered = filtered.filter(r => r.date <= endDate);
      if (filterBU) filtered = filtered.filter(r => r.bu === filterBU);
      if (contractor) filtered = filtered.filter(r => r.contractor === contractor);

      // Summary Cards Data
      document.getElementById('statTotal').textContent = filtered.length;
      const avgDepth = filtered.length ? (filtered.reduce((acc, r) => acc + (r.depth || 0), 0) / filtered.length).toFixed(2) : 0;
      const avgPenetro = filtered.length ? (filtered.reduce((acc, r) => acc + (r.penetrometer || 0), 0) / filtered.length).toFixed(2) : 0;
      const uniqueContractors = new Set(filtered.map(r => r.contractor)).size;

      document.getElementById('statDepth').textContent = `${avgDepth} cm`;
      document.getElementById('statPenetro').textContent = avgPenetro;
      document.getElementById('statContractors').textContent = uniqueContractors;

      // 1. GRAFIK SEPARATE KHUSUS BU (DESAIN MODERN BAR CHART)
      const buList = ['PT ITCI HUTANI MANUNGGAL', 'PT ADINDO HUTANI LESTARI', 'PT NUSANTARA KALIMANTAN LESTARI'];
      const buCounts = buList.map(bu => filtered.filter(r => r.bu === bu).length);

      const ctxBU = document.getElementById('chartBU').getContext('2d');
      if (chartBUInstance) chartBUInstance.destroy();

      chartBUInstance = new Chart(ctxBU, {
        type: 'bar',
        data: {
          labels: buList,
          datasets: [{
            label: 'Total Laporan',

              '#7e22ce',
              '#be185d',
              '#1d4ed8'
            ],
            borderWidth: 2,
            borderRadius: 8,
            borderSkipped: false,
            barThickness: 42

        },

          responsive: true,
          maintainAspectRatio: false,

            legend: { display: false },
            tooltip: {
              backgroundColor: 'rgba(17, 24, 39, 0.9)',
              titleFont: { size: 13, weight: 'bold' },
              bodyFont: { size: 12 },
              padding: 10,
              cornerRadius: 8

            datalabels: {
              anchor: 'end',
              align: 'top',
              font: { weight: 'bold', size: 11 },
              color: '#374151',
              formatter: Math.round

            y: {

      let c2m = 0, c25m = 0, c3m = 0;
      
      filtered.forEach(r => {
        if (r.contractor) contractorMap[r.contractor] = (contractorMap[r.contractor] || 0) + 1;
        if (r.spasing2m && r.spasing2m !== '-') c2m++;
        if (r.spasing25m && r.spasing25m !== '-') c25m++;
        if (r.spasing3m && r.spasing3m !== '-') c3m++;
      });

      const topContractors = Object.keys(contractorMap).slice(0, 4);
      const topContractorCounts = topContractors.map(c => contractorMap[c]);

      const barDataValues = [c2m, c25m, c3m, ...topContractorCounts];

      const depthLineValues = combinedLabels.map(() => avgDepth);
      const penetroLineValues = combinedLabels.map(() => avgPenetro);

      const ctxCombined = document.getElementById('chartCombined').getContext('2d');
      if (chartCombinedInstance) chartCombinedInstance.destroy();

      chartCombinedInstance = new Chart(ctxCombined, {
        type: 'bar',
        data: {
          labels: combinedLabels,
          datasets: [
            {
              type: 'bar',
              label: 'Jumlah Terisi / Frekuensi',
              data: barDataValues,
              backgroundColor: [
                'rgba(34, 197, 94, 0.85)',
                'rgba(59, 130, 246, 0.85)',
                'rgba(245, 158, 11, 0.85)',
                'rgba(20, 184, 166, 0.85)',
                'rgba(99, 102, 241, 0.85)',
                'rgba(168, 85, 247, 0.85)',
                'rgba(236, 72, 153, 0.85)'
              ],
              borderRadius: 6,
              yAxisID: 'yBar'
            },
            {
              type: 'line',
              label: 'Rata-rata Depth (cm)',
              data: depthLineValues,
              borderColor: '#2563eb',
              backgroundColor: '#2563eb',
              borderWidth: 2.5,
              tension: 0.3,
              pointRadius: 4,
              yAxisID: 'yLine'
            },
            {
              type: 'line',
              label: 'Rata-rata Penetrometer',
              data: penetroLineValues,
              borderColor: '#d97706',
              backgroundColor: '#d97706',
              borderWidth: 2.5,
              borderDash: [5, 5],
              tension: 0.3,
              pointRadius: 4,
              yAxisID: 'yLine'
            }
          ]
        },
        options: {
          responsive: true,
          maintainAspectRatio: false,
          plugins: {
            legend: {
              display: true,
              position: 'top',
              labels: { font: { size: 11, weight: '600' }, usePointStyle: true }
            },
            tooltip: {
              backgroundColor: 'rgba(17, 24, 39, 0.9)',
              padding: 10,
              cornerRadius: 8
            },
            datalabels: { display: false }
          },
          scales: {
            yBar: {
              type: 'linear',
              position: 'left',

    function exportToCSV() {
      if (reports.length === 0) return alert('Tidak ada data untuk diunduh.');
      const headers = ['Tanggal', 'Pelapor', 'BU', 'Sector', 'Kontraktor', 'Kode Unit', 'Operator', 'Jenis Pekerjaan', 'Spasing 2m', 'Spasing 2.5m', 'Spasing 3m', 'Depth (cm)', 'Penetrometer', 'Stoking'];
      const rows = reports.map(r => [
        `"${r.date}"`, `"${r.pelapor}"`, `"${r.bu}"`, `"${r.sector}"`, `"${r.contractor}"`, `"${r.unitCode}"`, `"${r.operator}"`,
        `"${r.jobType}"`, `"${r.spasing2m}"`, `"${r.spasing25m}"`, `"${r.spasing3m}"`, r.depth, r.penetrometer, `"${r.stoking}"`
      ]);

      const csvContent = 'data:text/csv;charset=utf-8,' + [headers.join(','), ...rows.map(e => e.join(','))].join('\n');
      const encodedUri = encodeURI(csvContent);
      const link = document.createElement('a');
      link.setAttribute('href', encodedUri);
      link.setAttribute('download', `Rekap_Spot_Check_MSC_${new Date().toISOString().slice(0,10)}.csv`);
      document.body.appendChild(link);
      link.click();
      document.body.removeChild(link);
    }

    function exportToJSON() {
      if (reports.length === 0) return alert('Tidak ada data untuk diunduh.');
      const dataStr = 'data:text/json;charset=utf-8,' + encodeURIComponent(JSON.stringify(reports, null, 2));
      const link = document.createElement('a');
      link.setAttribute('href', dataStr);
      link.setAttribute('download', `Backup_MSC_Data_${new Date().toISOString().slice(0,10)}.json`);
      document.body.appendChild(link);
      link.click();
      document.body.removeChild(link);
    }

    function importData(event) {
      const file = event.target.files[0];
      if (!file) return;

      const reader = new FileReader();
      reader.onload = function(e) {
        try {
          const imported = JSON.parse(e.target.result);
          if (Array.isArray(imported)) {
            reports = [...reports, ...imported];
            saveReportsToStorage();
            renderTable();
            populateContractorFilter();
            alert('Data berhasil diimpor!');
          } else {
            alert('Format JSON tidak valid.');
          }
        } catch (err) {
          alert('Gagal membaca file impor. Pastikan format file sesuai.');
        }
      };
      reader.readAsText(file);
    }
